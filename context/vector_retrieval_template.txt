import numpy as np
from sentence_transformers import SentenceTransformer
from typing import List, Dict, Any

def compute_cosine_similarity(vec1: np.ndarray, vec2: np.ndarray) -> float:
    """
    Computes the cosine similarity between two vectors[cite: 10].
    """
    dot_product = np.dot(vec1, vec2)
    norm1 = np.linalg.norm(vec1)
    norm2 = np.linalg.norm(vec2)
    
    if norm1 == 0 or norm2 == 0:
        return 0.0
        
    return float(dot_product / (norm1 * norm2))

class LocalCorrectionRetriever:
    """
    Handles local embedding generation and retrieval for document corrections[cite: 11].
    """
    def __init__(self, model_name: str):
        self.model = SentenceTransformer(model_name, device="cpu")
        self.passage_embeddings = None
        self.passages = []

    def encode(self, texts: List[str]) -> np.ndarray:
        """
        Encodes a list of text strings into dense numpy embeddings[cite: 11].
        """
        return self.model.encode(texts, convert_to_numpy=True)

    def index_past_corrections(self, corrections: List[Dict[str, str]]) -> None:
        """
        Indexes past human corrections for retrieval[cite: 11].
        Expects dictionaries with 'text_slice' and 'metadata'.
        """
        self.passages = corrections
        texts_to_encode = [item["text_slice"] for item in corrections]
        self.passage_embeddings = self.encode(texts_to_encode)

    def retrieve_closest_match(self, query_text: str) -> Dict[str, Any]:
        """
        Finds the single closest past correction using cosine similarity[cite: 10, 11].
        """
        if self.passage_embeddings is None or len(self.passages) == 0:
            raise ValueError("No corrections indexed.")

        query_embedding = self.encode([query_text])[0]
        
        best_score = 0.0
        best_match = None

        for idx, passage_emb in enumerate(self.passage_embeddings):
            score = compute_cosine_similarity(query_embedding, passage_emb)
            if score > best_score:
                best_score = score
                best_match = self.passages[idx]

        return {
            "match": best_match,
            "score": best_score
        }
